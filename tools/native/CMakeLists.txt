cmake_minimum_required(VERSION 3.24)
project(wemakemodel_expert_research LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE_DIR)
    message(FATAL_ERROR "Use prepare.py with a pinned llama.cpp source checkout")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" upstream)
add_executable(llama-expert-research expert_research.cpp)
target_link_libraries(llama-expert-research PRIVATE llama llama-common vendor::nlohmann)
target_compile_features(llama-expert-research PRIVATE cxx_std_17)
target_compile_definitions(llama-expert-research PRIVATE RESEARCH_LLAMA_COMMIT="10d4846a572aebf21c0482d140b00e74c0f1ad20")
set_target_properties(llama-expert-research PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
